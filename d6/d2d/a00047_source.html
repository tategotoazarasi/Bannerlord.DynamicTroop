72 _lastMainPartyAtSea = mainPartyAtSea;

122 private bool MainPartyTroopCompositionChanged(MobileParty mainParty) {
123 var currentTroopCounts = new Dictionary<CharacterObject, int>();
124
125 foreach (var rosterElement in mainParty.MemberRoster.GetTroopRoster()) {
126 if (rosterElement.Character is CharacterObject { IsHero: false } character && rosterElement.Number > 0)
127 currentTroopCounts[character] = rosterElement.Number;
128 }
129
130 if (currentTroopCounts.Count == _lastMainPartyTroopCounts.Count) {
131 var compositionChanged = false;
132 foreach (var troopCount in currentTroopCounts) {
133 if (!_lastMainPartyTroopCounts.TryGetValue(troopCount.Key, out var previousCount) || previousCount != troopCount.Value) {
134 compositionChanged = true;
135 break;
136 }
137 }
138
139 if (!compositionChanged)
140 return false;
141 }
142
143 _lastMainPartyTroopCounts.Clear();
144 foreach (var troopCount in currentTroopCounts)
145 _lastMainPartyTroopCounts.Add(troopCount.Key, troopCount.Value);
146
147 return true;
148 }

150 private static bool CanOpenArmory(MapBarVM mapBar, MobileParty? mainParty) {
151 return Campaign.Current != null &&
152 mainParty != null &&
153 Hero.MainHero is { IsPrisoner: false } &&
154 mapBar.IsEnabled &&
155 mapBar.MapTimeControl is { IsInMap: true, IsCenterPanelEnabled: true } &&
156 !mainParty.IsTransitionInProgress &&
157 mainParty.MapEvent == null &&
158 PlayerEncounter.Current == null;
159 }
