477 var familyType = kvp.Key;

486

496
497 if (harnessCnt == 0 && harnessIndex < harnesses.Length) { harnessIndex++; }
498 }
499 }

502 _horseAndHarnesses.Sort((x, y) => y.CompareTo(x));

609 private void AssignExtraEquipment(EquipmentFilter equipmentFilter, AssignmentFilter assignmentFilter) {
610 var candidates = new List<KeyValuePair<EquipmentElement, int>>();
611
612 foreach (var kv in _equipmentToAssign) {
613 if (kv.Key is { IsEmpty: false, Item: not null } &&
614 kv.Value > 0 &&
615 equipmentFilter(kv)) { candidates.Add(kv); }
616 }
617
618 if (candidates.Count == 0)
619 return;
620
621 candidates.Sort(static (a, b) => {
622 var tierCompare = b.Key.Item.Tier.CompareTo(a.Key.Item.Tier);
623 return tierCompare != 0 ? tierCompare : b.Key.Item.Value.CompareTo(a.Key.Item.Value);
624 });
625
626 var currentIndex = 0;
627
628 foreach (var assignment in Assignments) {
629 if (!assignmentFilter(assignment))
630 continue;
631
632 var slot = assignment.EmptyWeaponSlot;
633 if (!slot.HasValue)
634 continue;
635
636 while (currentIndex < candidates.Count && candidates[currentIndex].Value <= 0)
637 currentIndex++;
638
639 if (currentIndex >= candidates.Count)
640 return;
641
642 var current = candidates[currentIndex];
643
644 assignment.SetEquipment(slot.Value, current.Key);
645 if (ShouldWriteDistributionLog(Level.Debug))
646 WriteDistributionLog($"extra equipment {current.Key} assigned to {assignment.Character.StringId}#{assignment.Index} on slot {slot.Value} for {_party.Name}", Colors.Green, Level.Debug);
647
648 candidates[currentIndex] = new KeyValuePair<EquipmentElement, int>(current.Key, current.Value - 1);
649 ConsumeEquipmentToAssign(current.Key);
650 }

785 private void AssignEquipmentType(ItemTypeEnum type) {
786 var armours = _equipmentToAssign
787 .Where(kv => kv.Key is { IsEmpty: false, Item: not null } &&
788 kv.Value > 0 &&
789 kv.Key.Item.ItemType == type)
790 .Select(kv => new KeyValuePair<EquipmentElement, int>(kv.Key, kv.Value))
791 .ToArray();
792
793 var settings = ModSettings.Instance;
794 var preferDefaultEquipmentThenClosest = settings?.PreferDefaultEquipmentThenClosest ?? true;
795
796 var slotIndex = Global.GetEquipmentIndexByItemType(type);
797 if (slotIndex == null) { return; }
798
799 for (var assignmentIndex = 0; assignmentIndex < Assignments.Count; assignmentIndex++)
800 {
801 var assignment = Assignments[assignmentIndex];
802 var referenceElement = assignment.ReferenceEquipment.GetEquipmentFromSlot(slotIndex.Value);
803 var referenceItem = referenceElement.Item;
804
805 // Loyal ON: first try exact vanilla equipment.
806 if (preferDefaultEquipmentThenClosest && referenceItem != null)
807 {
808 for (var i = 0; i < armours.Length; i++)
809 {
810 var candidate = armours[i];
811 if (candidate.Value <= 0) { continue; }
812 if (candidate.Key.Item != referenceItem) { continue; }
813
814 assignment.SetEquipment(slotIndex.Value, candidate.Key);
815 armours[i] = new KeyValuePair<EquipmentElement, int>(candidate.Key, candidate.Value - 1);
816 ConsumeEquipmentToAssign(candidate.Key);
817 break;
818 }
819 }
820
821 if (!assignment.Equipment.GetEquipmentFromSlot(slotIndex.Value).IsEmpty) { continue; }
822
823 var maxAllowedTier = GetMaxAllowedTier(referenceItem);
824 var desiredCulture = assignment.Character.Culture;
825
826 // Most-compatible selection (Effectiveness proximity), still capped by +2 tier.
827 if (referenceItem != null)
828 {
829 var targetEffectiveness = referenceItem.Effectiveness;
830
831 var bestCultureIndex = -1;
832 var bestCultureDiff = float.MaxValue;
833 var bestCultureEffectiveness = float.MinValue;
834
835 var bestAnyIndex = -1;
836 var bestAnyDiff = float.MaxValue;
837 var bestAnyEffectiveness = float.MinValue;
838
839 for (var i = 0; i < armours.Length; i++)
840 {
841 var candidate = armours[i];
842 if (candidate.Value <= 0) { continue; }
843
844 var candidateItem = candidate.Key.Item;
845 if (candidateItem == null) { continue; }
846
847 if ((int)candidateItem.Tier > maxAllowedTier) { continue; }
848
849 var diff = Math.Abs(candidateItem.Effectiveness - targetEffectiveness);
850 var effectiveness = candidateItem.Effectiveness;
851
852 // ANY culture
853 if (diff < bestAnyDiff ||
854 diff == bestAnyDiff && effectiveness > bestAnyEffectiveness)
855 {
856 bestAnyIndex = i;
857 bestAnyDiff = diff;
858 bestAnyEffectiveness = effectiveness;
859 }
860
861 // CULTURE-compatible
862 if (!IsCultureCompatible(candidateItem, desiredCulture)) { continue; }
863
864 if (diff < bestCultureDiff ||
865 diff == bestCultureDiff && effectiveness > bestCultureEffectiveness)
866 {
867 bestCultureIndex = i;
868 bestCultureDiff = diff;
869 bestCultureEffectiveness = effectiveness;
870 }
871 }
872
873 var chosenIndex = bestAnyIndex;
874
875 if (preferDefaultEquipmentThenClosest && bestCultureIndex >= 0)
876 {
877 chosenIndex = bestCultureIndex;
878
879 if (bestAnyIndex >= 0)
880 {
881 // diff smaller = more compatible
882 var anyIsMoreThanTwiceAsCompatible = bestCultureDiff > (bestAnyDiff * CULTURE_OVERRIDE_COMPATIBILITY_MULTIPLIER);
883 if (anyIsMoreThanTwiceAsCompatible)
884 {
885 chosenIndex = bestAnyIndex;
886 }
887 }
888 }
889
890
891 if (chosenIndex >= 0)
892 {
893 var chosen = armours[chosenIndex];
894 assignment.SetEquipment(slotIndex.Value, chosen.Key);
895 armours[chosenIndex] = new KeyValuePair<EquipmentElement, int>(chosen.Key, chosen.Value - 1);
896 ConsumeEquipmentToAssign(chosen.Key);
897 }
898
899 continue;
900 }
901
902 // No reference item:
903 {
904 var bestCultureIndex = -1;
905 var bestCultureEffectiveness = float.MinValue;
906
907 var bestAnyIndex = -1;
908 var bestAnyEffectiveness = float.MinValue;
909
910 for (var i = 0; i < armours.Length; i++)
911 {
912 var candidate = armours[i];
913 if (candidate.Value <= 0) { continue; }
914
915 var candidateItem = candidate.Key.Item;
916 if (candidateItem == null) { continue; }
917
918 if ((int)candidateItem.Tier > maxAllowedTier) { continue; }
919
920 var effectiveness = candidateItem.Effectiveness;
921
922 // ANY
923 if (effectiveness > bestAnyEffectiveness)
924 {
925 bestAnyIndex = i;
926 bestAnyEffectiveness = effectiveness;
927 }
928
929 // CULTURE-compatible
930 if (!IsCultureCompatible(candidateItem, desiredCulture)) { continue; }
931
932 if (effectiveness > bestCultureEffectiveness)
933 {
934 bestCultureIndex = i;
935 bestCultureEffectiveness = effectiveness;
936 }
937 }
938
939 var chosenIndex =
940 preferDefaultEquipmentThenClosest
941 ? (bestCultureIndex >= 0 ? bestCultureIndex : bestAnyIndex)
942 : bestAnyIndex;
943
944 if (chosenIndex >= 0)
945 {
946 var chosen = armours[chosenIndex];
947 assignment.SetEquipment(slotIndex.Value, chosen.Key);
948 armours[chosenIndex] = new KeyValuePair<EquipmentElement, int>(chosen.Key, chosen.Value - 1);
949 ConsumeEquipmentToAssign(chosen.Key);
950 }
951 }
952 }

1187 private bool IsWeaponSuitable(EquipmentElement equipment, ItemObject referenceWeapon, Assignment assignment, bool strict) {
1188 if (equipment.IsEmpty ||
1189 equipment.Item == null ||
1190 !equipment.Item.HasWeaponComponent ||
1191 _equipmentToAssign[equipment] <= 0 ||
1192 !equipment.Item.IsSuitableForCharacter(assignment.Character) ||
1193 !Global.HaveSameWeaponClass(Global.GetWeaponClass(equipment.Item), Global.GetWeaponClass(referenceWeapon))) { return false; }
1194
1195 var isSuitableForMount = equipment.Item.IsSuitableForMount();
1196 var isCouchable = equipment.Item.IsCouchable();
1197
1198 if (strict) {
1199 if (!Global.FullySameWeaponClass(equipment.Item, referenceWeapon)) { return false; }
1200
1201 if (assignment.IsMounted) {
1202 // 严格模式下骑马：必须适合骑乘；如果是长杆武器，则必须可进行骑枪冲刺
1203 return isSuitableForMount && (!equipment.Item.IsPolearm() || isCouchable);
1204 }
1205
1206 // 严格模式下非骑马：不可选择可进行骑枪冲刺的武器
1207 return !isCouchable;
1208 }
1209
1210 // 非严格模式下：骑马的不能选择不适合骑乘的武器，非骑马的可以选择任意武器
1211 return !assignment.IsMounted || isSuitableForMount;
