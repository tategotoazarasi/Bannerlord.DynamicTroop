166 private void GarbageCollectEquipments(MobileParty mobileParty) {
167 if (!mobileParty.IsValid() || mobileParty.MemberRoster?.GetTroopRoster() == null) return;
168
169 var memberCnt = mobileParty.MemberRoster.GetTroopRoster()
170 .WhereQ(element => element.Character is { IsHero: false })
171 .SumQ(element => element.Number);
172 foreach (var equipmentAndThreshold in EquipmentAndThresholds) {
173 var partyArmory = PartyArmories.GetValueSafe(mobileParty.Id);
174 if (partyArmory == null || partyArmory.IsEmpty()) return;
175
176 var armorTotalCount = partyArmory.WhereQ(kv => kv.Key.ItemType == equipmentAndThreshold.Key)
177 .SumQ(kv => kv.Value);
178 var surplusCount = armorTotalCount - memberCnt;
179 if (surplusCount <= 0) continue;
180
181 var surplusCountCpy = surplusCount;
182
183 // 创建优先级队列
184 ItemPriorityQueue armorQueue = new(new EquipmentEffectivenessComparer());
185 foreach (var kv in partyArmory.WhereQ(kv => kv.Key.ItemType == equipmentAndThreshold.Key))
186 armorQueue.Enqueue(kv.Key, ((int)kv.Key.Tier, kv.Key.Value));
187
188 // 移除多余的装备
189 while (surplusCount > 0 && armorQueue.Count > 0) {
190 var lowestArmor = armorQueue.Dequeue();
191 var countToRemove = Math.Min(partyArmory[lowestArmor.Key], surplusCount);
192 partyArmory[lowestArmor.Key] -= countToRemove;
193 surplusCount -= countToRemove;
194 if (partyArmory[lowestArmor.Key] == 0) _ = partyArmory.Remove(lowestArmor.Key);
195 }
196
197 Global.Debug($"Garbage collected {surplusCountCpy - surplusCount}x{equipmentAndThreshold.Key} armors from party {mobileParty.Name}");
198 }

361 private void DistributePlayerSimulationLoot(
362 MapEvent mapEvent,
363 MBReadOnlyList<MapEventParty> winnerParties,
364 MBReadOnlyList<MapEventParty> loserParties) {
365 var random = new Random();
366 var dropRate = SubModule.Settings?.DropRate ?? 1f;
367 var canUseMountEquipment = !mapEvent.IsNavalMapEvent && !MapEventHelper.IsNavalRaid(mapEvent);
368 var mountsUnavailable = !canUseMountEquipment ||
369 mapEvent.IsSiegeAssault ||
370 mapEvent.IsSiegeOutside ||
371 mapEvent.IsSallyOut;
372
373 foreach (var defeatedParty in loserParties) {
374 Dictionary<UniqueTroopDescriptor, Assignment>? assignments = null;
375 var defeatedMobileParty = defeatedParty.Party.MobileParty;
376 if (defeatedMobileParty != null && defeatedMobileParty.IsValid()) {
377 var defeatedArmory = SanitizePartyArmory(defeatedMobileParty.Id);
378 if (defeatedArmory != null)
379 assignments = PartyEquipmentDistributor.CreateMapEventAssignments(
380 defeatedMobileParty,
381 defeatedParty.Troops,
382 defeatedArmory,
383 canUseMountEquipment);
384 }
385
386 var lootChances = Campaign.Current.Models.BattleRewardModel
387 .GetLootCasualtyChances(winnerParties, defeatedParty.Party);
388
389 foreach (var casualty in defeatedParty.Troops) {
390 if (casualty.Troop.IsHero ||
391 casualty.State is not (RosterTroopState.Killed or RosterTroopState.WoundedInThisBattle))
392 continue;
393
394 var totalChance = 0f;
395 foreach (var lootChance in lootChances) {
396 var mobileParty = lootChance.Key.Party.MobileParty;
397 if (lootChance.Value > 0f &&
398 (lootChance.Key.Party == PartyBase.MainParty ||
399 mobileParty != null && mobileParty.IsValid() && PartyArmories.ContainsKey(mobileParty.Id)))
400 totalChance += lootChance.Value;
401 }
402
403 if (totalChance <= 0f)
404 continue;
405
406 var roll = random.NextDouble() * totalChance;
407 var accumulatedChance = 0f;
408 MapEventParty? recipientParty = null;
409 foreach (var lootChance in lootChances) {
410 var mobileParty = lootChance.Key.Party.MobileParty;
411 if (lootChance.Value <= 0f ||
412 (lootChance.Key.Party != PartyBase.MainParty &&
413 (mobileParty == null || !mobileParty.IsValid() || !PartyArmories.ContainsKey(mobileParty.Id))))
414 continue;
415
416 accumulatedChance += lootChance.Value;
417 if (roll < accumulatedChance) {
418 recipientParty = lootChance.Key;
419 break;
420 }
421 }
422
423 if (recipientParty == null)
424 continue;
425
426 Assignment? assignment = null;
427 Equipment equipment;
428 if (assignments != null && assignments.TryGetValue(casualty.Descriptor, out assignment))
429 equipment = assignment.Equipment;
430 else
431 equipment = casualty.Troop.RandomBattleEquipment;
432
433 ItemObject? extraBrokenArmor = null;
434 var extraBreakChance = casualty.Troop.Tier switch {
435 3 => 0.25f,
436 4 => 0.35f,
437 5 => 0.45f,
438 _ => casualty.Troop.Tier >= 6 ? 0.60f : 0f
439 };
440
441 if (extraBreakChance > 0f && random.NextFloat() <= extraBreakChance) {
442 var armorCandidates = new List<ItemObject>();
443 foreach (var armorSlot in Global.ArmourAndHorsesSlots) {
444 var armor = equipment[armorSlot].Item;
445 if (armor is { HasArmorComponent: true } && armor.ItemType != ItemObject.ItemTypeEnum.HorseHarness)
446 armorCandidates.Add(armor);
447 }
448
449 if (armorCandidates.Count > 0)
450 extraBrokenArmor = armorCandidates[random.Next(armorCandidates.Count)];
451 }
452
453 // auto resolve dont have damages per body part
454 foreach (var slot in Global.EquipmentSlots) {
455 if (mountsUnavailable && (slot == EquipmentIndex.Horse || slot == EquipmentIndex.HorseHarness))
456 continue;
457
458 var equipmentElement = equipment[slot];
459 var item = equipmentElement.Item;
460 if (item == null ||
461 item.IsBannerItem ||
462 item.ItemType == ItemObject.ItemTypeEnum.Banner ||
463 assignment?.IsTemporarySlot(slot, item) == true ||
464 (extraBrokenArmor != null && item.StringId == extraBrokenArmor.StringId) ||
465 !ItemBlackList.Test(item) ||
466 random.NextFloat() > dropRate)
467 continue;
468
469 if (recipientParty.Party == PartyBase.MainParty)
470 ArmyArmory.AddItemToArmory(item);
471 else if (recipientParty.Party.MobileParty is { } recipientMobileParty)
472 AddItemToPartyArmory(recipientMobileParty.Id, item, 1);
473 }
474 }
475 }

498 private void DistributeLootRandomly(MapEventParty[] winnerParties,
499 float totalWinnerStrength,
500 IEnumerable<MapEventParty> loserParties) {
501 var lootItemsWithCount = GetAllLootItems(loserParties);
502
503 if (winnerParties.Length == 1) {
504 var winnerParty = winnerParties[0].Party.MobileParty;
505 var totalLootCount = 0;
506
507 foreach (var lootItem in lootItemsWithCount) {
508 if (winnerParty == MobileParty.MainParty)
509 ArmyArmory.AddItemToArmory(lootItem.Key, lootItem.Value);
510 else
511 AddItemToPartyArmory(winnerParty.Id, lootItem.Key, lootItem.Value);
512
513 totalLootCount += lootItem.Value;
514 }
515
516 if (totalLootCount > 0)
517 Global.Debug($"Party {winnerParty.Id} looted {totalLootCount} items");
518
519 return;
520 }
521
522 Random random = new();
523 Dictionary<MBGUID, int> partyLootCount = new();
524 Dictionary<MBGUID, int> currentItemDistribution = new();
525
526 foreach (var lootItem in lootItemsWithCount) {
527 var item = lootItem.Key;
528 var itemCount = lootItem.Value;
529 var selectionFailed = false;
530 currentItemDistribution.Clear();
531
532 for (var i = 0; i < itemCount; i++) {
533 var chosenParty = ChoosePartyRandomly(winnerParties, totalWinnerStrength, random);
534 if (chosenParty == null) {
535 selectionFailed = true;
536 break;
537 }
538
539 currentItemDistribution[chosenParty.Id] =
540 currentItemDistribution.TryGetValue(chosenParty.Id, out var currentItemCount)
541 ? currentItemCount + 1
542 : 1;
543 partyLootCount[chosenParty.Id] =
544 partyLootCount.TryGetValue(chosenParty.Id, out var currentPartyCount)
545 ? currentPartyCount + 1
546 : 1;
547 }
548
549 foreach (var partyItemCount in currentItemDistribution) {
550 if (partyItemCount.Key == MobileParty.MainParty.Id)
551 ArmyArmory.AddItemToArmory(item, partyItemCount.Value);
552 else
553 AddItemToPartyArmory(partyItemCount.Key, item, partyItemCount.Value);
554 }
555
556 if (selectionFailed)
557 return;
558 }
559
560 // 记录每个party获得的物品总数
561 foreach (var partyCount in partyLootCount)
562 Global.Debug($"Party {partyCount.Key} looted {partyCount.Value} items");
563 }

565 private static Dictionary<ItemObject, int> GetAllLootItems(IEnumerable<MapEventParty> parties) {
566 Dictionary<ItemObject, int> itemsWithCount = new();
567 foreach (var party in parties) {
568 var mobileParty = party.Party.MobileParty;
569 if (mobileParty != null) {
570 var inventory = SanitizePartyArmory(mobileParty.Id);
571 if (inventory != null) {
572 foreach (var item in inventory) {
573 if (item.Value <= 0 || !ItemBlackList.Test(item.Key))
574 continue;
575
576 itemsWithCount[item.Key] = itemsWithCount.TryGetValue(item.Key, out var currentCount)
577 ? currentCount + item.Value
578 : item.Value;
579 }
580
581 continue;
582 }
583 }
584 }
585
586 return itemsWithCount;
587 }

645 public static Dictionary<ItemObject, int>? SanitizePartyArmory(MBGUID partyId) {
646 if (!PartyArmories.TryGetValue(partyId, out var armory))
647 return null;
648
649 var requiresRebuild = false;
650 var normalizedItems = new HashSet<ItemObject>();
651
652 foreach (var entry in armory) {
653 if (entry.Value <= 0 ||
654 !ArmyArmory.TryResolveArmoryItem(entry.Key, out var item) ||
655 !ReferenceEquals(item, entry.Key) ||
656 !normalizedItems.Add(item)) {
657 requiresRebuild = true;
658 break;
659 }
660 }
661
662 if (!requiresRebuild)
663 return armory;
664 var sanitizedItems = new Dictionary<ItemObject, int>();
665 foreach (var entry in armory) {
666 if (entry.Value <= 0 || !ArmyArmory.TryResolveArmoryItem(entry.Key, out var item))

737 var partyId = new MBGUID(pendingParty.Key);
